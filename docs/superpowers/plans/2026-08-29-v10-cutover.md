# v10 Cutover Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Zatrzymać zachowaną flotę v9 i uruchomić od zera pełną flotę Rocky Linux 10 na osobnych VMID-ach i adresach.

**Architecture:** v9 pozostaje w repozytorium i na PVE jako zatrzymany rollback. v10 dostaje osobną platformę `xenonv10` oraz dwóch najemców `orionv10-r10` i `cassiopeiav10-r10`; wszystkie VM-y są tworzone przez wspólny moduł `pve_vm_set`, z trzema niezależnymi rootami Terraform. Stare i nowe definicje nie współdzielą adresów własnych, VMID-ów, hostgroupów ani kont aplikacyjnych.

**Tech Stack:** Terraform 1.15.6, provider `bpg/proxmox` 0.111.1, Proxmox VE API, Ansible, Rocky Linux 10.2, MariaDB/Galera, ProxySQL, Keepalived, PMM, MinIO, S3, Pythonowe sondy repozytorium.

## Global Constraints

- Pracuj na gałęzi `feat/v10-cutover`; nie przełączaj ani nie resetuj `audit/t3-judo`.
- Nie zmieniaj żadnego pliku v8/v9; katalogi i VM-y v9 są rollbackiem.
- Użyj dokładnie obrazu `local:import/Rocky-10.2-GenericCloud.qcow2` oraz istniejącego snippet-u `local:snippets/r10-cloud-init.yaml`.
- Użyj `versions/versions-el10.lock.yml` i `versions.policy: locked`.
- Stan Terraform jest lokalny i ignorowany przez Git; każdy root inicjalizuj bez `-upgrade`.
- Każde `terraform apply` wykonuj serialnie przez `-parallelism=1`; przed apply przejrzyj zapisany plan.
- Nie wykonuj `terraform destroy` dla v9 i nie stosuj automatycznego hard-stopu po timeoutcie graceful shutdown.
- Nie migruj danych v9; v10 jest czystą instalacją.
- Nie wypisuj wartości sekretów ani pełnych odpowiedzi API; środowisko ładuj z `tests/lab/.env` bez echo tokenów.

Create `/tmp/isa-v10-graceful-shutdown.py` with this exact runner and execute it only after the two backup probes pass:

```python
import json
import os
import ssl
import sys
import time
from pathlib import Path
from urllib.parse import quote
from urllib.request import Request, urlopen

import yaml

ENDPOINT = os.environ["PROXMOX_VE_ENDPOINT"].rstrip("/")
TOKEN = os.environ["PROXMOX_VE_API_TOKEN"]
POOL = "claude-isa"
INVENTORIES = {
    "orion": (Path("clusters/orionv9-r9/inventory.yml"), ("galera", "restore")),
    "cassiopeia": (Path("clusters/cassiopeiav9-r9/inventory.yml"), ("galera", "restore")),
    "platform": (Path("platform/xenonv9/inventory.yml"), ("proxysql", "app", "infra")),
}
TLS = ssl.create_default_context()
TLS.check_hostname = False
TLS.verify_mode = ssl.CERT_NONE


def pve(path, method="GET"):
    request = Request(
        ENDPOINT + "/api2/json" + path,
        data=b"" if method == "POST" else None,
        method=method,
        headers={"Authorization": f"PVEAPIToken={TOKEN}"},
    )
    with urlopen(request, context=TLS, timeout=30) as response:
        return json.load(response)["data"]


def inventory_hosts(path, groups):
    data = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
    children = (data.get("all") or {}).get("children") or {}
    result = []
    for group in groups:
        hosts = (children.get(group) or {}).get("hosts") or {}
        for host in hosts:
            if host not in result:
                result.append(host)
    return result


def fail(message):
    raise RuntimeError(message)


def wait_task(node, upid):
    deadline = time.monotonic() + 180
    task_path = (
        f"/nodes/{quote(str(node), safe='')}/tasks/"
        f"{quote(str(upid), safe='')}/status"
    )
    while time.monotonic() < deadline:
        status = pve(task_path)
        if status.get("status") == "stopped":
            if status.get("exitstatus") != "OK":
                fail(f"task {upid} zakończony: {status.get('exitstatus')}")
            return
        time.sleep(2)
    fail(f"timeout task {upid}")


def wait_vm_stopped(node, vmid):
    deadline = time.monotonic() + 180
    vm_path = f"/nodes/{quote(str(node), safe='')}/qemu/{vmid}/status/current"
    while time.monotonic() < deadline:
        status = pve(vm_path)
        if status.get("status") == "stopped":
            return
        time.sleep(2)
    fail(f"timeout VMID {vmid}: status nie jest stopped")



def main():
    tenant_hosts = (
        inventory_hosts(*INVENTORIES["orion"])
        + inventory_hosts(*INVENTORIES["cassiopeia"])
    )
    platform_hosts = inventory_hosts(*INVENTORIES["platform"])
    ordered_hosts = tenant_hosts + platform_hosts
    if len(ordered_hosts) != 12 or len(set(ordered_hosts)) != 12:
        fail(f"inventory v9 nie daje dokładnie 12 unikalnych hostów: {ordered_hosts}")

    resources = pve("/cluster/resources?type=vm")
    by_name = {
        str(item.get("name")): item
        for item in resources
        if item.get("pool") == POOL
    }
    missing = sorted(set(ordered_hosts) - set(by_name))
    if missing:
        fail(f"brak VM-ów v9 w puli {POOL}: {missing}")
    not_running = sorted(
        name for name in ordered_hosts if by_name[name].get("status") != "running"
    )
    if not_running:
        fail(f"v9 nie jest w całości running przed shutdownem: {not_running}")

    for name in ordered_hosts:
        item = by_name[name]
        node = str(item.get("node") or "")
        if not node:
            fail(f"brak noda PVE dla {name} VMID {item.get('vmid')}")
        vmid = int(item["vmid"])
        upid = pve(
            f"/nodes/{quote(node, safe='')}/qemu/{vmid}/status/shutdown",
            method="POST",
        )
        if not upid:
            fail(f"brak UPID dla graceful shutdown {name} VMID {vmid}")
        print(f"shutdown {name} vmid={vmid} node={node}")
        wait_task(node, upid)
        wait_vm_stopped(node, vmid)
        print(f"stopped {name} vmid={vmid} node={node}")


try:
    main()
except Exception as exc:
    print(f"FAIL: {type(exc).__name__}: {str(exc)[:200]}", file=sys.stderr)
    sys.exit(1)
```
